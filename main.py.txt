import os
import random
import requests
from datetime import datetime
from generate_image import create_quote_image

# 🔐 البيانات من GitHub Secrets
BOT_TOKEN = os.getenv("BOT_TOKEN")
CHANNEL_ID = "@BlackWidowHouse"  # قناتك العامة
WEBSITE_URL = "https://dftr5ana.netlify.app/?utm_source=telegram&utm_medium=bot&utm_campaign=daily_quote"

def load_quotes(file="quotes.txt"):
    """تحميل الاقتباسات من الملف"""
    if not os.path.exists(file):
        print(f"⚠️ ملف {file} غير موجود!")
        return []
    with open(file, "r", encoding="utf-8") as f:
        lines = [line.strip() for line in f if line.strip() and "|" in line]
    return [line.split("|", 1) for line in lines]

def send_to_telegram(image_path, quote, author):
    """إرسال الصورة والاقتباس للقناة"""
    url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendPhoto"
    
    caption = (
        f"📖 *اقتباس اليوم من دفتر خانة*\n\n"
        f"🔹 \"{quote}\"\n"
        f"— {author}\n\n"
        f"🌐 اكتشف آلاف الكتب والاقتباسات: {WEBSITE_URL}\n"
        f"#اقتباسات #دفتر_خانة #قراءة #أدب #هشام_أفندي"
    )
    
    with open(image_path, "rb") as img:
        payload = {
            "chat_id": CHANNEL_ID,
            "caption": caption,
            "parse_mode": "Markdown"
        }
        files = {"photo": img}
        res = requests.post(url, data=payload, files=files)
        
    if res.status_code == 200:
        print(f"✅ تم النشر بنجاح | {datetime.now().strftime('%Y-%m-%d %H:%M')}")
        return True
    else:
        print(f"❌ فشل النشر: {res.text}")
        return False

def main():
    print("🤖 بدء بوت دفتر خانة للاقتباسات اليومية...")
    
    quotes = load_quotes()
    if not quotes:
        print("⚠️ لا توجد اقتباسات للتحميل!")
        return
        
    quote, author = random.choice(quotes)
    img_path = "daily_quote.jpg"
    
    print(f"🎨 جاري توليد الصورة لاقتباس: {quote[:30]}...")
    create_quote_image(quote, author, img_path)
    
    if os.path.exists(img_path):
        print("📡 جاري الإرسال للقناة @BlackWidowHouse...")
        send_to_telegram(img_path, quote, author)
        # تنظيف الملف بعد الإرسال
        if os.path.exists(img_path):
            os.remove(img_path)
    else:
        print("❌ فشل إنشاء الصورة!")

if __name__ == "__main__":
    main()
